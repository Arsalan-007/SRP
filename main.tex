\section{Problem Statement}
% ---------------------------------------------------

\begin{frame}{The Challenge: Why Do Trees Still Win?}
    \begin{columns}[c]
        \column{0.48\textwidth}
        \textbf{Tree models are strong on tabular data because they:}
        \vspace{0.3cm}
        \begin{itemize}
            \setlength\itemsep{0.3em}
            \item work directly with individual features
            \item naturally model threshold-like decisions
            \item benefit from ensembles of many learners
        \end{itemize}

        \column{0.48\textwidth}
        \begin{block}{Our Core Questions}
            \begin{enumerate}
                \item Can a neural ensemble genuinely close this gap?
                \item Can we do it, without losing full end-to-end differentiability?
            \end{enumerate}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{Our Approach: Bagged Expert Ensembles (BEE)}
    \begin{columns}[c]
        \column{0.55\textwidth}
        \textbf{The Pipeline Flow:}
        \vspace{0.3cm}

        \textbf{Feature bagging:} encourage different experts to learn different information
        \vspace{0.25cm}

        \textbf{Feature embeddings:} process each feature separately before they are mixed
        \vspace{0.25cm}

        \textbf{Aggregation:} combine the expert representations
        \vspace{0.25cm}

        \textbf{Key Early Finding:} Mean pooling remained a strong and simple baseline across our experiments.

        \column{0.45\textwidth}
        \centering
        \includegraphics[width=\linewidth]{figures/pipeline_flow.png}
    \end{columns}
\end{frame}

\section{Questions We Investigated}
% ---------------------------------------------------

\begin{frame}{Question 1: Are the Experts Learning the Same Thing?}
    \begin{columns}[t]
        \column{0.5\textwidth}
        \large
        Without Feature Bagging \\
        Expert 1 $\rightarrow$ A B C D E \\
        Expert 2 $\rightarrow$ A B C D E \\
        Expert 3 $\rightarrow$ A B C D E

        \column{0.5\textwidth}
        \large
        With Feature Bagging \\
        Expert 1 $\rightarrow$ A B D \\
        Expert 2 $\rightarrow$ B C E \\
        Expert 3 $\rightarrow$ A D E
    \end{columns}
    \vspace{0.4cm}
    \large Each expert sees $\sim$70\% of the features.
    \vspace{0.3cm}
    \begin{block}{Verified, Mean off-diagonal cosine similarity: 0.108}
        Lower similarity suggests the learned expert representations are not identical.
    \end{block}
\end{frame}

\begin{frame}{Question 2: Does the Model Depend on the Original Feature Axes?}
    \textbf{What is Rotation Invariance?} \\
    In tabular data, features have distinct, independent meanings (e.g., Age vs.\ Income). Rotation
    mathematically scrambles these features together. A model is ``rotation-invariant'' if it treats the
    distinct data and the scrambled data exactly the same.
    \vspace{0.5cm}

    \textbf{The ``17-Point'' Diagnostic Test -- Poker Hand}\,{\scriptsize[5]} \\
    Rotating input data destroys XGBoost's accuracy (e.g., a 17-point drop) because trees require unmixed
    axes to make splits. However, it cost our un-embedded MLP essentially nothing. This proved our neural
    network was natively blurring features together.
\end{frame}

\begin{frame}{Question 2: Does the Model Depend on the Original Feature Axes?}
    \begin{center}
        \includegraphics[width=0.72\linewidth]{figures/rotation_invariance.pdf}
    \end{center}
    \vspace{0.2cm}
    \begin{center}
        \large \textbf{The Fix: PLR} tokenizes each feature independently before they ever interact.
    \end{center}
\end{frame}

\begin{frame}{PLR: Processing Each Feature Before Mixing}
    \begin{columns}[c]
        \column{0.62\textwidth}
        \textbf{"PLR" = Periodic $\rightarrow$ Linear $\rightarrow$ ReLU} \textit{(Gorishniy et al., 2022)}\,{\scriptsize[2]}
        \vspace{0.3cm}
        \begin{enumerate}
            \item \textbf{PERIODIC:} Map value into $\sin/\cos$ waves at \textit{learned} frequencies.
            \item \textbf{LINEAR:} Shrink to a small per-feature vector.
            \item \textbf{ReLU:} Standard non-linearity.
        \end{enumerate}

        \column{0.35\textwidth}
        \centering
        \includegraphics[height=0.5\textheight]{figures/plr_flow.png}
    \end{columns}
    \vspace{0.2cm}
    \begin{block}{Why it Works}
        PLR takes one numerical feature and turns it into a small nonlinear vector before any feature mixing happens.
    \end{block}
\end{frame}

\begin{frame}{Question 3: Does Attention Earn Its Place?}
    \small
    \textbf{Test:} 8 benchmark datasets, same experts, same size ($\sim$38k parameters), same training.
    The only difference is how expert opinions are combined: \textbf{simply average them}, or \textbf{causal attention}.
    \vspace{0.1cm}
    \begin{table}
        \centering
        \renewcommand{\arraystretch}{1.12}
        \footnotesize
        \begin{tabular}{l | c | c | l}
            \rowcolor{PrimaryColor} \color{white}\textbf{Dataset} & \color{white}\textbf{Averaging} & \color{white}\textbf{Causal attention} & \color{white}\textbf{Better} \\
            California Housing (RMSE$\downarrow$)$^{*}$ & 0.5404 & \textbf{0.4985} & attention \\
            \rowcolor{LightGrey} Adult (accuracy$\uparrow$) & \textbf{0.8589} & 0.8567 & averaging \\
            Jannis (accuracy$\uparrow$) & \textbf{0.7195} & 0.7097 & averaging \\
            \rowcolor{LightGrey} Covertype (accuracy$\uparrow$) & \textbf{0.9324} & 0.9100 & averaging \\
            YearPredictionMSD (RMSE$\downarrow$) & \textbf{8.8293} & 8.8472 & averaging (within noise) \\
            \rowcolor{LightGrey} Higgs Small (accuracy$\uparrow$) & \textbf{0.7215} & 0.7173 & averaging \\
            Facebook Comments (RMSE$\downarrow$) & \textbf{5.9443} & 6.0738 & averaging \\
            \rowcolor{LightGrey} Santander (accuracy$\uparrow$) & \textbf{0.9147} & 0.9131 & averaging \\
        \end{tabular}
    \end{table}
    \vspace{-0.1cm}
    \textbf{Result:} averaging wins on \textbf{7 of 8} datasets, and attention takes 2--3$\times$ longer to train.
    \textbf{So we simply average.}\\[0.1cm]
    \tiny $^{*}$These runs were capped at 60 epochs, which cut averaging short on California Housing. In our follow-up
    test trained to convergence (5 seeds each), averaging wins there too.
\end{frame}

\section{Introduction to Negative Correlation Learning}
% ---------------------------------------------------

\begin{frame}{Negative Correlation Learning (NCL)}
    \textbf{Idea}\,{\scriptsize[6]}\textbf{:} Instead of forcing diversity architecturally (feature bagging alone), penalize learners for making the \textit{same mistakes}, directly in the loss function.

    \begin{equation*}
        \mathcal{L}_{i} = \text{MSE}(F_{i}, y) - \lambda(F_{i} - \overline{F})^{2}
    \end{equation*}

    \begin{itemize}
        \item \textbf{Accuracy} is rewarded (MSE term).
        \item \textbf{Distance} from the ensemble mean is rewarded too (penalty term)—pulling learners apart to cover blind spots.
        \item Requires moving pooling from embedding-space to output-space, so each learner has its own scalar prediction to penalize.
    \end{itemize}
\end{frame}

\begin{frame}{The Strongest Result: NCL on California Housing}
    \begin{columns}
        \column{0.52\textwidth}
        \centering
        \includegraphics[width=\linewidth]{figures/ncl_california_housing.pdf}
        \column{0.48\textwidth}
        \begin{table}
            \centering
            \renewcommand{\arraystretch}{1.2}
            \scriptsize
            \begin{tabular}{l | c}
                \rowcolor{PrimaryColor} \color{white}\textbf{Model} & \color{white}\textbf{RMSE} \\
                TabR (2023, single)\,{\scriptsize[3]} & 0.4010 \\
                \rowcolor{SecondaryColor} \color{white}\textbf{BEE $\lambda$=1.0} & \color{white}\textbf{0.4039} \\
                Without NCL & 0.4202 \\
                \rowcolor{LightGrey} TabM (2025, single)\,{\scriptsize[4]} & 0.4292 \\
                Our XGBoost (tuned) & 0.4328 \\
                \rowcolor{LightGrey} Pub. FT-Transformer\,{\scriptsize[1]} & 0.4590 \\
            \end{tabular}
        \end{table}
        \small
        Beats our tuned XGBoost, published FT-Transformer, \textbf{and TabM (2025)} — within 0.003 of TabR, the closest our model has come to the current state of the art. Not even jointly tuned with $\lambda$: the architecture was kept fixed and only $\lambda$ was varied.
    \end{columns}
\end{frame}

\begin{frame}{NCL Results: Does It Generalize Beyond California Housing?}
    \begin{table}
        \centering
        \renewcommand{\arraystretch}{1.35}
        \resizebox{\textwidth}{!}{%
        \begin{tabular}{l | l | c | c | l}
            \rowcolor{PrimaryColor} \color{white}\textbf{Dataset} & \color{white}\textbf{Task} & \color{white}\textbf{Without NCL} & \color{white}\textbf{With NCL} & \color{white}\textbf{Effect} \\
            California Housing (RMSE $\downarrow$) & Regression & 0.4202 & \textbf{0.4039} ($\lambda=1.0$) & \cellcolor{WinGreen}\checkmark\ helps (beats XGBoost) \\
            Facebook (RMSE $\downarrow$) & Regression & 5.826 & \textbf{5.673} ($\lambda=0.8$) & \cellcolor{WinGreen}\checkmark\ helps \\
            Higgs Small (Acc $\uparrow$) & Classification & 0.7245 & $\approx$0.51 (all $\lambda>0$) & \cellcolor{LossRed}$\times$\ collapses to one class \\
            Santander (Acc $\uparrow$) & Classification & 0.9218 & 0.8995 (all $\lambda>0$) & \cellcolor{LossRed}$\times$\ predicts majority class \\
        \end{tabular}}
    \end{table}
    \vspace{0.3cm}
    \textbf{Bottom line:} NCL helps on regression (2 of 2) but collapses on classification (0 of 2).\\[0.2cm]
    \small
    \textbf{Why classification breaks:} the penalty pushes on unbounded logits, so the loss loses its lower
    bound and training blows up. Fixing this (probability-space NCL) is our first next step.
\end{frame}

\section{Conclusion \& Next Steps}
% ---------------------------------------------------

\begin{frame}{Scorecard \& Limitations}
    \textbf{Main Findings}
    \vspace{0.3cm}
    \begin{itemize}
        \setlength\itemsep{0.4em}
        \item \textbf{Feature Bagging -- Useful} \\ Different feature subsets encourage expert diversity.
        \item \textbf{Feature Embeddings - Useful, but dataset dependent} \\ PLR helps in several cases but not universally.
        \item \textbf{Attention - No consistent advantage} \\ Mean pooling is simpler and avoids expert-order dependence.
    \end{itemize}
    \vspace{0.4cm}
    \textbf{Main limitation:} \\
    Covertype still shows a significant performance gap.
\end{frame}
